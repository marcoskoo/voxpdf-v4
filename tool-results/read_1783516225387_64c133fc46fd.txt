  1203→/* ══ SENTENCE / WORD BUILDER ══
  1204→   Segmenta un párrafo en oraciones y palabras.
  1205→   Cada oración queda en un <span class="sent-sp">
  1206→   con las palabras dentro como <span class="word-sp">.
  1207→   Devuelve {el, wEls, sEls, sentOffsets}:
  1208→     wEls        — todos los span de palabra (para compatibilidad)
  1209→     sEls        — los span de oración
  1210→     sentOffsets — posición de char inicio de cada oración en el texto del párrafo

  1212→function _splitSentences(text){
  1213→  const re=/[^.!?;]+[.!?;]*/g;
  1214→  const sents=[];let m;
  1215→  while((m=re.exec(text))!==null){if(m[0].trim())sents.push({text:m[0],offset:m.index});}
  1216→  if(!sents.length)sents.push({text,offset:0});
  1217→  return sents;
  1218→}

  1320→/* hiSentence — resaltado sincronizado de oración + palabra.
  1321→   Usa busqueda binaria sobre wOffsets (offsets exactos grabados en data-off)
  1322→   para localizar la palabra activa en O(log n) sin acumulacion de chars.
  1323→   charIdx es el charIndex del evento onboundary, relativo al bigText global.
  1324→   localChar = charIdx - S.offsets[ri]  =>  posicion dentro del parrafo. */
  1325→function hiSentence(ri, charIdx){
  1326→  const pi=S.readList[ri];
  1327→  const para=S.paras[pi];if(!para)return;
  1328→  const off=S.offsets[ri];
  1329→  const localChar=charIdx-off;   // posicion en el texto del parrafo
  1330→
  1331→  // — Limpiar estado anterior de este parrafo —
  1332→  para.sEls?.forEach(s=>s.classList.remove('sent-hi'));
  1333→  para.wEls?.forEach(w=>w.classList.remove('hi'));
  1334→
  1335→  // — Oración activa: busqueda lineal inversa sobre sentOffsets (pocas oraciones) —
  1336→  const sos=para.sentOffsets||[];
  1337→  let sIdx=0;
  1338→  for(let i=sos.length-1;i>=0;i--){
  1339→    if(sos[i]<=localChar){sIdx=i;break;}
  1340→  }
  1341→  para.sEls?.[sIdx]?.classList.add('sent-hi');
  1342→
  1343→  // — Palabra activa: busqueda binaria sobre wOffsets (exacto, sin acumulacion) —
  1344→  const wo=para.wOffsets;
  1345→  const wels=para.wEls;
  1346→  if(!wo||!wo.length)return;
  1347→  // Encontrar el ultimo wOffset <= localChar
  1348→  let lo=0,hi=wo.length-1,found=-1;
  1349→  while(lo<=hi){
  1350→    const mid=(lo+hi)>>1;
  1351→    if(wo[mid]<=localChar){found=mid;lo=mid+1;}
  1352→    else hi=mid-1;
  1353→  }
  1354→  if(found>=0){
  1355→    // Verificar que localChar esta dentro de la longitud de la palabra
  1356→    const wEl=wels[found];
  1357→    const wStart=wo[found];
  1358→    const wEnd=wStart+wEl.textContent.length;
  1359→    if(localChar>=wStart&&localChar<=wEnd+1){
  1360→      wEl.classList.add('hi');
  1361→    }else if(found+1<wels.length){
  1362→      // El TTS puede adelantar el cursor — resaltar la siguiente si hay duda
  1363→      wels[found+1]?.classList.add('hi');
  1364→    }
  1365→  }
  1366→}
