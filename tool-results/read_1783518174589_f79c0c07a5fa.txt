    61→function splitSentences(text: string): { text: string; offset: number }[] {
    62→  const re = /[^.!?;]+[.!?;]*/g;
    63→  const sents: { text: string; offset: number }[] = [];
    64→  let m;
    65→  while ((m = re.exec(text)) !== null) {
    66→    if (m[0].trim()) sents.push({ text: m[0], offset: m.index });
    67→  }
    68→  if (!sents.length) sents.push({ text, offset: 0 });
    69→  return sents;
    70→}

    72→function detectLang(text: string): string {
    73→  const sample = text.slice(0, 200);
    74→  if (/[àáâãäåèéêëìíîïòóôõöùúûüýÿñç]/i.test(sample)) return 'es';
    75→  if (/[äöüß]/i.test(sample)) return 'de';
    76→  if (/[àâçéèêëîïôùûü]/i.test(sample)) return 'fr';
    77→  return 'en';
    78→}

   305→  async function loadPDF(file: File) {
   306→    if (!pdfjsLib) {
   307→      toast({ title: 'Cargando PDF.js...', description: 'Espera un momento' });
   308→      await new Promise(r => setTimeout(r, 2000));
   309→      if (!pdfjsLib) throw new Error('PDF.js no disponible');
   310→    }
   311→    const ab = await file.arrayBuffer();
   312→    // Try Web Worker first for non-blocking parsing
   313→    try {
   314→      const worker = new Worker(new URL('@/workers/pdf-worker.ts', import.meta.url));
   315→      const result = await new Promise<any>((resolve, reject) => {
   316→        worker.onmessage = (e) => {
   317→          if (e.data.type === 'done') { resolve(e.data.data); worker.terminate(); }
   318→          else if (e.data.type === 'error') { reject(new Error(e.data.error)); worker.terminate(); }
   319→        };
   320→        worker.postMessage({ type: 'parse', data: ab }, [ab]);
   321→      });
   322→      store.setParagraphs(result.paragraphs);
   323→      store.setTotalPages(result.totalPages);
   324→    } catch {
   325→      // Fallback to main thread
   326→      const doc = await pdfjsLib.getDocument({ data: ab }).promise;
   327→      store.setTotalPages(doc.numPages);
   328→      await extractRenderPDF(doc);
   329→    }
   330→    addRecent(file.name, 0);
   331→    restoreProgress(file.name);
   332→    buildTOC();
   333→    computeWordFrequency();
   334→    computeMindMap();
   335→    toast({ title: file.name, description: `PDF cargado` });
   336→  }

   338→  async function extractRenderPDF(pdfDoc: any) {
   339→    const paras: any[] = [];
   340→    const N = pdfDoc.numPages;
   341→    // Lazy rendering: only extract visible pages first
   342→    const batchSize = store.lazyRendering ? 5 : N;
   343→    for (let pg = 1; pg <= Math.min(batchSize, N); pg++) {
   344→      const page = await pdfDoc.getPage(pg);
   345→      const content = await page.getTextContent();
   346→      const vp = page.getViewport({ scale: 1 });
   347→      const pH = vp.height;
   348→      const HZ = pH * 0.08;
   349→      const FZ = pH * 0.08;
   350→      const items = content.items.slice().sort((a: any, b: any) => b.transform[5] - a.transform[5]);
   351→      const lines: any[] = [];
   352→      let cur: any[] = [];
   353→      let cy: number | null = null;
   354→      for (const it of items) {
   355→        const y = it.transform[5];
   356→        if (cy === null || Math.abs(y - cy) < 3) { cur.push(it); cy = y; }
   357→        else { if (cur.length) lines.push({ y: cy, items: cur }); cur = [it]; cy = y; }
   358→      }
   359→      if (cur.length) lines.push({ y: cy, items: cur });
   360→      const pageText = lines.map((l: any) => l.items.map((i: any) => i.str).join('')).join('').trim();
   361→      if (!pageText) continue; // OCR would go here for scanned pages
   362→      let buf: string[] = [];
   363→      let lastT: string | null = null;
   364→      const flush = (isH: boolean, isF: boolean) => {
   365→        const txt = buf.join(' ').replace(/\s+/g, ' ').trim();
   366→        buf = [];
   367→        if (!txt) return;
   368→        paras.push({ text: txt, page: pg, isHeader: isH, isFooter: isF });
   369→      };
   370→      for (const l of lines) {
   371→        const txt = l.items.map((i: any) => i.str).join(' ').trim();
   372→        if (!txt) continue;
   373→        const relY = pH - l.y;
   374→        const t = relY < HZ ? 'h' : relY > (pH - FZ) ? 'f' : 'b';
   375→        if (lastT && t !== lastT) flush(lastT === 'h', lastT === 'f');
   376→        buf.push(txt);
   377→        lastT = t;
   378→      }
   379→      if (buf.length) flush(lastT === 'h', lastT === 'f');
   380→    }
   381→    store.setParagraphs(paras);
   382→  }

   384→  async function loadEPUB(file: File) {
   385→    // Dynamic JSZip import
   386→    const JSZip = (await import('jszip')).default;
   387→    const ab = await file.arrayBuffer();
   388→    const zip = await JSZip.loadAsync(ab);
   389→    const containerXml = await zip.file('META-INF/container.xml')?.async('text');
   390→    const opfPath = containerXml?.match(/full-path="([^"]+\.opf)"/i)?.[1];
   391→    if (!opfPath) throw new Error('EPUB sin OPF');
   392→    const opfBase = opfPath.substring(0, opfPath.lastIndexOf('/') + 1);
   393→    const opfDoc = new DOMParser().parseFromString(await zip.file(opfPath)?.async('text') || '', 'application/xml');
   394→    const manifest: Record<string, string> = {};
   395→    opfDoc.querySelectorAll('manifest item').forEach((item: any) => {
   396→      manifest[item.id] = opfBase + item.getAttribute('href');
   397→    });
   398→    const spine = Array.from(opfDoc.querySelectorAll('spine itemref')).map((i: any) => i.getAttribute('idref'));
   399→    let allText = '';
   400→    for (const id of spine) {
   401→      const href = manifest[id];
   402→      if (!href) continue;
   403→      const f = zip.file(href);
   404→      if (!f) continue;
   405→      const html = await f.async('text');
   406→      const hdoc = new DOMParser().parseFromString(html, 'text/html');
   407→      hdoc.querySelectorAll('script,style,nav').forEach(e => e.remove());
   408→      allText += (hdoc.body?.textContent || '') + '\n\n';
   409→    }
   410→    renderPlainText(allText);
   411→    addRecent(file.name, 0);
   412→    toast({ title: file.name, description: 'EPUB cargado' });
   413→  }

   519→  // ── TOC ──
   520→  function buildTOC() {
   521→    const chapters: any[] = [];
   522→    const reT = /^(cap[ií]tulo|chapter|parte|part|secci[oó]n|section|art[ií]culo|tema|unidad|ap[eé]ndice|introduction|introducci[oó]n|conclusion|resumen|summary|abstract|prefacio|pr[oó]logo)\s*[\d.\-:IVXivx]*/i;
   523→    const reN = /^(\d+[.\-)]|[IVXivx]+[.\-)]|\([a-z\d]+\))\s+\S/;
   524→    store.paragraphs.forEach((p, i) => {
   525→      if (p.isHeader || p.isFooter) return;
   526→      const w = p.text.trim().split(/\s+/);
   527→      if (w.length > 14) return;
   528→      const t = p.text.trim();
   529→      if (reT.test(t) || reN.test(t) || (w.length <= 10 && t.length > 4)) {
   530→        chapters.push({ title: t.slice(0, 60), startIdx: i, page: p.page });
   531→      }
   532→    });
   533→    if (chapters.length < 2) {
   534→      const byPage: any = {};
   535→      store.paragraphs.forEach((p, i) => {
   536→        if (!byPage[p.page]) byPage[p.page] = { page: p.page, startIdx: i };
   537→      });
   538→      const pages = Object.values(byPage).sort((a: any, b: any) => a.page - b.page);
   539→      if (pages.length <= 30) {
   540→        pages.forEach((pg: any) => chapters.push({ title: `Página ${pg.page}`, startIdx: pg.startIdx, page: pg.page }));
   541→      }
   542→    }
   543→    store.setChapters(chapters);
   544→  }

   556→  // ── Mind Map ──
   557→  function computeMindMap() {
   558→    const root: MindMapNode = { id: 'root', label: store.fileName || 'Documento', children: [], depth: 0 };
   559→    store.chapters.forEach((ch, i) => {
   560→      const chapterNode: MindMapNode = { id: `ch-${i}`, label: ch.title, children: [], depth: 1 };
   561→      // Get first few paragraphs of chapter as sub-topics
   562→      const start = ch.startIdx;
   563→      const end = i < store.chapters.length - 1 ? store.chapters[i + 1].startIdx : Math.min(start + 5, store.paragraphs.length);
   564→      for (let j = start; j < Math.min(start + 3, end); j++) {
   565→        const firstSentence = store.paragraphs[j]?.text.split(/[.!?]/)[0] || '';
   566→        if (firstSentence.length > 5) {
   567→          chapterNode.children.push({
   568→            id: `ch-${i}-p${j}`,
   569→            label: firstSentence.slice(0, 60),
   570→            children: [],
   571→            depth: 2,
   572→          });
   573→        }
   574→      }
   575→      root.children.push(chapterNode);
   576→    });
   577→    if (root.children.length === 0 && store.paragraphs.length > 0) {
   578→      // Fallback: group by pages
   579→      const byPage: Record<number, string[]> = {};
   580→      store.paragraphs.forEach(p => {
   581→        if (!byPage[p.page]) byPage[p.page] = [];
   582→        byPage[p.page].push(p.text.split(/[.!?]/)[0]);
   583→      });
   584→      Object.entries(byPage).slice(0, 15).forEach(([pg, texts]) => {
   585→        root.children.push({
   586→          id: `pg-${pg}`,
   587→          label: `Página ${pg}`,
   588→          children: texts.slice(0, 2).map((t, i) => ({
   589→            id: `pg-${pg}-t${i}`, label: t.slice(0, 60), children: [], depth: 2
   590→          })),
   591→          depth: 1,
   592→        });
   593→      });
   594→    }
   595→    store.setMindMap(root);
   596→  }

   598→  // ── TTS Engine ──
   599→  function startReading(fromIdx?: number) {
   600→    stopReading(true);
   601→    const startIdx = fromIdx ?? store.currentParaIdx;
   602→    const readableParas = store.paragraphs
   603→      .map((p, i) => ({ ...p, idx: i }))
   604→      .filter(p => !(p.isHeader || p.isFooter) || !store.skipHF)
   605→      .filter(p => p.idx >= startIdx);
   606→    if (!readableParas.length) {
   607→      toast({ title: 'Fin del documento' });
   608→      return;
   609→    }
   610→
   611→    // Pre-synthesis lookahead: split into sentence chunks
   612→    if (store.pauseOnPunctuation) {
   613→      readWithPunctuationPauses(readableParas, 0);
   614→    } else {
   615→      readContinuous(readableParas);
   616→    }
   617→  }

   619→  function readContinuous(paras: any[]) {
   620→    const bigText = paras.map(p => p.text).join(' \n ');
   621→    const utt = new SpeechSynthesisUtterance(bigText);
   622→    const voice = store.voices[store.selectedVoice]?.voice;
   623→    if (voice) utt.voice = voice;
   624→    utt.rate = store.rate;
   625→    utt.pitch = store.pitch;
   626→    utt.volume = store.volume;
   627→    utt.lang = voice?.lang || 'es-ES';
   628→
   629→    utt.onboundary = (e) => {
   630→      if (!store.playing) return;
   631→      // Find current paragraph by char index
   632→      let charCount = 0;
   633→      for (const p of paras) {
   634→        charCount += p.text.length + 3;
   635→        if (charCount > e.charIndex) {
   636→          store.setCurrentParaIdx(p.idx);
   637→          saveProgress(store.fileName, p.idx);
   638→          scrollToPara(p.idx);
   639→          break;
   640→        }
   641→      }
   642→    };
   643→
   644→    utt.onend = () => {
   645→      store.setPlaying(false);
   646→      releaseWakeLock();
   647→    };
   648→
   649→    utteranceRef.current = utt;
   650→    store.setPlaying(true);
   651→    acquireWakeLock();
   652→    speechSynthesis.speak(utt);
   653→  }

   655→  function readWithPunctuationPauses(paras: any[], paraOffset: number) {
   656→    if (paraOffset >= paras.length) {
   657→      store.setPlaying(false);
   658→      releaseWakeLock();
   659→      return;
   660→    }
   661→    const p = paras[paraOffset];
   662→    const sents = splitSentences(p.text);
   663→
   664→    store.setCurrentParaIdx(p.idx);
   665→    scrollToPara(p.idx);
   666→    saveProgress(store.fileName, p.idx);
   667→
   668→    let sentIdx = 0;
   669→    function speakNext() {
   670→      if (sentIdx >= sents.length) {
   671→        // Move to next paragraph
   672→        readWithPunctuationPauses(paras, paraOffset + 1);
   673→        return;
   674→      }
   675→      const sent = sents[sentIdx];
   676→      // Auto language switch
   677→      let lang = store.voices[store.selectedVoice]?.voice?.lang || 'es-ES';
   678→      if (store.autoLanguageSwitch) {
   679→        const detected = detectLang(sent.text);
   680→        const matchingVoice = store.voices.findIndex(v => v.lang.startsWith(detected));
   681→        if (matchingVoice >= 0) lang = store.voices[matchingVoice].voice.lang;
   682→      }
   683→      const utt = new SpeechSynthesisUtterance(sent.text);
   684→      const voice = store.voices[store.selectedVoice]?.voice;
   685→      if (voice) utt.voice = voice;
   686→      utt.rate = store.rate;
   687→      utt.pitch = store.pitch;
   688→      utt.volume = store.volume;
   689→      utt.lang = lang;
   690→
   691→      // Pre-synthesis lookahead: prepare next utterance while current speaks
   692→      utt.onend = () => {
   693→        sentIdx++;
   694→        // Pause at punctuation marks
   695→        const lastChar = sent.text.trim().slice(-1);
   696→        const pauseMs = lastChar === '.' ? 500 : lastChar === ',' ? 200 : lastChar === ';' ? 350 : lastChar === '!' || lastChar === '?' ? 400 : 100;
   697→        setTimeout(speakNext, pauseMs);
   698→      };
   699→
   700→      utteranceRef.current = utt;
   701→      speechSynthesis.speak(utt);
   702→    }
   703→
   704→    store.setPlaying(true);
   705→    acquireWakeLock();
   706→    speakNext();
   707→  }

   765→  function setupEqualizer() {
   766→    if (!audioCtxRef.current) audioCtxRef.current = new AudioContext();
   767→    const ctx = audioCtxRef.current;
   768→    const frequencies = [60, 230, 910, 4000, 14000];
   769→    const filters = frequencies.map((freq, i) => {
   770→      const filter = ctx.createBiquadFilter();
   771→      filter.type = i === 0 ? 'lowshelf' : i === frequencies.length - 1 ? 'highshelf' : 'peaking';
   772→      filter.frequency.value = freq;
   773→      filter.gain.value = store.eqBands[i] || 0;
   774→      filter.Q.value = 1;
   775→      return filter;
   776→    });
   777→    // Chain filters
   778→    for (let i = 0; i < filters.length - 1; i++) {
   779→      filters[i].connect(filters[i + 1]);
   780→    }
   781→    filters[filters.length - 1].connect(ctx.destination);
   782→    eqFiltersRef.current = filters;
   783→  }

   953→  // ── Glossary ──
   954→  function handleTextSelection() {

   966→      setGlossaryPopupPos({ x: rect.left, y: rect.bottom });

   969→      // Offer to add to glossary
   970→      const range = sel.getRangeAt(0);
   971→      const rect = range.getBoundingClientRect();
