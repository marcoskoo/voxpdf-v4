async function extractRenderPDF(pdfDoc,filename){
  const viewer=document.getElementById('pdf-viewer');viewer.innerHTML='';S.paras=[];
  const N=pdfDoc.numPages;
  for(let pg=1;pg<=N;pg++){
    showLoading('Extrayendo… '+pg+'/'+N);
    const page=await pdfDoc.getPage(pg);
    const content=await page.getTextContent();
    const vp=page.getViewport({scale:1});
    const pH=vp.height,HZ=pH*.08,FZ=pH*.08;
    const items=content.items.slice().sort((a,b)=>b.transform[5]-a.transform[5]);
    const lines=[];let cur=[],cy=null;
    for(const it of items){const y=it.transform[5];if(cy===null||Math.abs(y-cy)<3){cur.push(it);cy=y;}else{if(cur.length)lines.push({y:cy,items:cur});cur=[it];cy=y;}}
    if(cur.length)lines.push({y:cy,items:cur});
    // Check if page has no text → offer OCR
    const pageText=lines.map(l=>l.items.map(i=>i.str).join('')).join('').trim();
    const pageDiv=document.createElement('div');pageDiv.className='pdf-page';
    const badge=document.createElement('div');badge.className='pg-badge';badge.textContent='Página '+pg+' de '+N;pageDiv.appendChild(badge);
    if(!pageText&&N>0){
      // Scanned page — render as image + run OCR
      const canvas=document.createElement('canvas');
      const vp2=page.getViewport({scale:1.5});
      canvas.width=vp2.width;canvas.height=vp2.height;
      await page.render({canvasContext:canvas.getContext('2d'),viewport:vp2}).promise;
      await _ocrCanvas(canvas,pg,pageDiv,filename);
    }else{
      let buf=[],lastT=null;
      const flush=(isH,isF)=>{
        const txt=buf.join(' ').replace(/\s+/g,' ').trim();buf=[];if(!txt)return;
        const idx=S.paras.length;
        const {el,wEls,wOffsets,sEls,sentOffsets}=_buildParaEl(txt,idx);
        if(isH||isF)el.classList.add('hf-line');
        if(S.highlights[idx])el.classList.add('highlighted');
        if(S.bookmarks[idx]){el.classList.add('bookmarked');_addBkDot(el);}
        if(isH||isF){const tg=document.createElement('span');tg.className='skip-tag';tg.textContent=isH?'encab.':'pie';el.appendChild(tg);}
        S.paras.push({text:txt,page:pg,isH,isF,el,wEls,wOffsets,sEls,sentOffsets});pageDiv.appendChild(el);
      };
      for(const l of lines){
        const txt=l.items.map(i=>i.str).join(' ').trim();if(!txt)continue;
        const relY=pH-l.y;const t=relY<HZ?'h':relY>(pH-FZ)?'f':'b';
        if(lastT&&t!==lastT)flush(lastT==='h',lastT==='f');buf.push(txt);lastT=t;
      }
      if(buf.length)flush(lastT==='h',lastT==='f');
    }
    viewer.appendChild(pageDiv);
  }
  applyHFvisibility();updateProgress();
}

/* ══ SENTENCE / WORD BUILDER ══
   Segmenta un párrafo en oraciones y palabras.
   Cada oración queda en un <span class="sent-sp">
   con las palabras dentro como <span class="word-sp">.
   Devuelve {el, wEls, sEls, sentOffsets}:
     wEls        — todos los span de palabra (para compatibilidad)
     sEls        — los span de oración
     sentOffsets — posición de char inicio de cada oración en el texto del párrafo
*/
function _splitSentences(text){
  const re=/[^.!?;]+[.!?;]*/g;
  const sents=[];let m;
  while((m=re.exec(text))!==null){if(m[0].trim())sents.push({text:m[0],offset:m.index});}
  if(!sents.length)sents.push({text,offset:0});
  return sents;
}
/* _buildParaEl — cada wEl guarda su offset exacto de char en data-off.
   Esto permite a hiSentence localizar la palabra activa con busqueda
   binaria en O(log n) sin acumulacion propensa a desalineamiento. */
function _buildParaEl(txt,idx){
  const el=document.createElement('p');
  el.className='pdf-para';el.dataset.idx=idx;
  el.setAttribute('role','paragraph');el.setAttribute('tabindex','0');
  el.addEventListener('contextmenu',e=>{e.preventDefault();openCtxMenu(e,idx);});
  el.addEventListener('click',()=>jumpTo(idx));
  el.addEventListener('keydown',e=>{if(e.key==='Enter')jumpTo(idx);if(e.key==='b'||e.key==='B')openNoteModal(idx);});
  const sents=_splitSentences(txt);
  const wEls=[];
  const wOffsets=[];   // offset de char de inicio de cada palabra en txt
  const sEls=[];
  const sentOffsets=[];
  sents.forEach(sent=>{
    sentOffsets.push(sent.offset);
    const sSpan=document.createElement('span');sSpan.className='sent-sp';
    // Tokenizar preservando posicion absoluta en txt
    const wordRe=/\S+/g;
    let wm;let lastEnd=sent.offset;
    while((wm=wordRe.exec(sent.text))!==null){
      const wordStart=sent.offset+wm.index;
      if(wordStart>lastEnd)sSpan.appendChild(document.createTextNode(txt.slice(lastEnd,wordStart)));
      const wSpan=document.createElement('span');
      wSpan.className='word-sp';
      wSpan.textContent=wm[0];
      wSpan.dataset.off=wordStart;   // offset exacto en txt
      sSpan.appendChild(wSpan);
      wEls.push(wSpan);
      wOffsets.push(wordStart);
      lastEnd=wordStart+wm[0].length;
    }
    const sentEnd=sent.offset+sent.text.length;
    if(lastEnd<sentEnd)sSpan.appendChild(document.createTextNode(txt.slice(lastEnd,sentEnd)));
    el.appendChild(sSpan);sEls.push(sSpan);
  });
  return{el,wEls,wOffsets,sEls,sentOffsets};
}

/* hiSentence — resaltado sincronizado de oración + palabra.
   Usa busqueda binaria sobre wOffsets (offsets exactos grabados en data-off)
   para localizar la palabra activa en O(log n) sin acumulacion de chars.
   charIdx es el charIndex del evento onboundary, relativo al bigText global.
   localChar = charIdx - S.offsets[ri]  =>  posicion dentro del parrafo. */
function hiSentence(ri, charIdx){
  const pi=S.readList[ri];
  const para=S.paras[pi];if(!para)return;
  const off=S.offsets[ri];
  const localChar=charIdx-off;   // posicion en el texto del parrafo

  // — Limpiar estado anterior de este parrafo —
  para.sEls?.forEach(s=>s.classList.remove('sent-hi'));
  para.wEls?.forEach(w=>w.classList.remove('hi'));

  // — Oración activa: busqueda lineal inversa sobre sentOffsets (pocas oraciones) —
  const sos=para.sentOffsets||[];
  let sIdx=0;
  for(let i=sos.length-1;i>=0;i--){
    if(sos[i]<=localChar){sIdx=i;break;}
  }
  para.sEls?.[sIdx]?.classList.add('sent-hi');

  // — Palabra activa: busqueda binaria sobre wOffsets (exacto, sin acumulacion) —
  const wo=para.wOffsets;
  const wels=para.wEls;
  if(!wo||!wo.length)return;
  // Encontrar el ultimo wOffset <= localChar
  let lo=0,hi=wo.length-1,found=-1;
  while(lo<=hi){
    const mid=(lo+hi)>>1;
    if(wo[mid]<=localChar){found=mid;lo=mid+1;}
    else hi=mid-1;
  }
  if(found>=0){
    // Verificar que localChar esta dentro de la longitud de la palabra
    const wEl=wels[found];
    const wStart=wo[found];
    const wEnd=wStart+wEl.textContent.length;
    if(localChar>=wStart&&localChar<=wEnd+1){
      wEl.classList.add('hi');
    }else if(found+1<wels.length){
      // El TTS puede adelantar el cursor — resaltar la siguiente si hay duda
      wels[found+1]?.classList.add('hi');
    }
  }
}

function detectChapters(){
  const chapters=[];
  const reT=/^(cap[ií]tulo|chapter|parte|part|secci[oó]n|section|art[ií]culo|tema|unidad|ap[eé]ndice|appendix|introduction|introducci[oó]n|conclusion|resumen|summary|abstract|prefacio|pr[oó]logo)\s*[\d.\-:IVXivx]*/i;
  const reN=/^(\d+[.\-)]|[IVXivx]+[.\-)]|\([a-z\d]+\))\s+\S/;
  S.paras.forEach((p,i)=>{
    if(p.isH||p.isF)return;const w=p.text.trim().split(/\s+/);if(w.length>14)return;
    const t=p.text.trim();const cr=t.replace(/\s/g,'').split('').filter(c=>/[A-ZÁÉÍÓÚÑÜ]/.test(c)).length/Math.max(t.replace(/\s/g,'').length,1);
    if(reT.test(t)||reN.test(t)||(cr>.65&&w.length<=10&&t.length>4)||(w.length<=5&&t.length>3&&/[A-ZÁÉÍÓÚÑÜ]/.test(t[0])&&!/[.,:;]$/.test(t)))
      chapters.push({title:t.slice(0,60)+(t.length>60?'…':''),startIdx:i,page:p.page});
  });
  if(chapters.length<2){
    const byPage={};S.paras.forEach((p,i)=>{if(!p.isH&&!p.isF&&!byPage[p.page])byPage[p.page]={page:p.page,startIdx:i};});
    const pages=Object.values(byPage).sort((a,b)=>a.page-b.page);chapters.length=0;
    if(pages.length<=60)pages.forEach(pg=>chapters.push({title:'Página '+pg.page,startIdx:pg.startIdx,page:pg.page}));
    else{const step=Math.ceil(pages.length/30);for(let i=0;i<pages.length;i+=step){const g=pages.slice(i,i+step);chapters.push({title:'Páginas '+g[0].page+'–'+g[g.length-1].page,startIdx:g[0].startIdx,page:g[0].page});}}
  }
  return chapters;
}

/* ══════════════════════════════════════════
   NUEVA FUNCIÓN 2: SPEED READING RSVP
   Muestra una palabra a la vez en el centro
   de la pantalla. Velocidad ajustable 100-800
   palabras por minuto. Resalta la letra pivote.
══════════════════════════════════════════ */
const RSVP={words:[],idx:0,wpm:300,playing:false,timer:null,startParaIdx:0};
function openRSVP(fromIdx){
  if(!S.paras.length){toast('Carga un archivo primero','err');return;}
  const start=fromIdx!==undefined?fromIdx:S.curIdx;
  RSVP.startParaIdx=start;
  // Build word list from current position
  const paras=buildReadList(start);
  RSVP.words=paras.flatMap(i=>S.paras[i].text.split(/\s+/).filter(w=>w.length>0));
  RSVP.idx=0;RSVP.playing=false;
  if(RSVP.timer){clearInterval(RSVP.timer);RSVP.timer=null;}
  document.getElementById('rsvp-overlay').classList.add('open');
  document.getElementById('rsvp-topbar-pill').classList.add('show');
  // Pause TTS if playing
  if(S.playing)pauseReading();
  _rsvpRender();
  _rsvpUpdateBtn();
}
function closeRSVP(){
  RSVP.playing=false;if(RSVP.timer){clearInterval(RSVP.timer);RSVP.timer=null;}
  document.getElementById('rsvp-overlay').classList.remove('open');
  document.getElementById('rsvp-topbar-pill').classList.remove('show');
}
function toggleRSVP(){if(RSVP.playing)_rsvpPause();else _rsvpPlay();}
function _rsvpPlay(){
  if(RSVP.idx>=RSVP.words.length){RSVP.idx=0;}
  RSVP.playing=true;_rsvpUpdateBtn();
  const ms=Math.round(60000/RSVP.wpm);
  RSVP.timer=setInterval(()=>{
    if(RSVP.idx>=RSVP.words.length){_rsvpPause();return;}
    _rsvpRender();RSVP.idx++;
    // Update progress bar
    document.getElementById('rsvp-pbar').style.width=(RSVP.idx/RSVP.words.length*100)+'%';
  },ms);
}
function _rsvpPause(){RSVP.playing=false;if(RSVP.timer){clearInterval(RSVP.timer);RSVP.timer=null;}_rsvpUpdateBtn();}
function _rsvpRender(){
  const word=RSVP.words[RSVP.idx]||'';
  // Find optimal pivot character (typically ~35% from start)
  const pivotPos=Math.max(1,Math.floor(word.length*.35));
  const before=word.slice(0,pivotPos-1);
  const pivot=word[pivotPos-1]||'';
  const after=word.slice(pivotPos);
  document.getElementById('rsvp-word').innerHTML=`${esc(before)}<span class="pivot">${esc(pivot)}</span>${esc(after)}`;
  setV('rsvp-meta',(RSVP.idx+1)+' / '+RSVP.words.length+' palabras');
}
function _rsvpUpdateBtn(){setV('rsvp-play-btn',RSVP.playing?'⏸ Pausar':'▶ Iniciar');}
function rsvpSetWPM(v){RSVP.wpm=v;setV('rsvp-wpm-val',v+' ppm');if(RSVP.playing){_rsvpPause();_rsvpPlay();}}
function rsvpSkip(n){
  _rsvpPause();
  RSVP.idx=Math.max(0,Math.min(RSVP.idx+n,RSVP.words.length-1));
  _rsvpRender();document.getElementById('rsvp-pbar').style.width=(RSVP.idx/RSVP.words.length*100)+'%';
}

function _drawStatsChart(buckets){
  const canvas=document.getElementById('stats-chart');if(!canvas)return;
  const dpr=window.devicePixelRatio||1;
  const W=canvas.parentElement.clientWidth-24||460;const H=140;
  canvas.width=W*dpr;canvas.height=H*dpr;canvas.style.width=W+'px';canvas.style.height=H+'px';
  const ctx=canvas.getContext('2d');ctx.scale(dpr,dpr);
  const maxSecs=Math.max(...buckets.map(b=>b.secs),1);
  const accent=getComputedStyle(document.documentElement).getPropertyValue('--accent').trim()||'#7c6af5';
  const text3=getComputedStyle(document.documentElement).getPropertyValue('--text3').trim()||'#5a5a70';

  ctx.clearRect(0,0,W,H);
  buckets.forEach((b,i)=>{
    const x=i*gap+(gap-barW)/2;const barH=Math.floor((b.secs/maxSecs)*(H-28));const y=H-barH-18;
    // Bar
    ctx.fillStyle=accent+'99';ctx.beginPath();ctx.roundRect?ctx.roundRect(x,y,barW,barH,3):ctx.rect(x,y,barW,barH);ctx.fill();

    // Label every 2nd day
    if(i%2===0){ctx.fillStyle=text3;ctx.font=`${8*dpr/dpr}px DM Mono,monospace`;ctx.textAlign='center';ctx.fillText(b.label,x+barW/2,H-4);}
  });

async function _elSynthesize(text){
  const model=document.getElementById('el-model')?.value||EL.model;
  const stability=parseFloat(document.getElementById('el-stability')?.value??EL.stability);
  const similarity=parseFloat(document.getElementById('el-similarity')?.value??EL.similarity);
  const style=parseFloat(document.getElementById('el-style')?.value??EL.style);
  const res=await fetch(`${EL_BASE}/text-to-speech/${EL.voiceId}`,{method:'POST',headers:{'xi-api-key':EL.apiKey,'Content-Type':'application/json','Accept':'audio/mpeg'},body:JSON.stringify({text,model_id:model,voice_settings:{stability,similarity_boost:similarity,style,use_speaker_boost:true}})});

function _elPlayBuffer(audioBuffer,paraIdx){
  return new Promise((resolve,reject)=>{
    if(!S.playing||!EL.isPlaying){reject(new Error('aborted'));return;}
    const src=EL.audioCtx.createBufferSource();src.buffer=audioBuffer;src.connect(EL.audioCtx.destination);EL.audioSource=src;
    const duration=audioBuffer.duration*1000;const words=S.paras[paraIdx].wEls||[];
    const sents=S.paras[paraIdx].sEls||[];
    if((words.length||sents.length)&&document.getElementById('hi1')?.checked){
      const interval=duration/Math.max(words.length,1);let wi=0;
      // También resaltar oraciones proporcionalmente
      const sentInterval=sents.length>1?duration/sents.length:duration;let si=0;
      const ht=setInterval(()=>{
        // Palabra
        if(wi>0)words[wi-1]?.classList.remove('hi');
        if(wi<words.length){words[wi]?.classList.add('hi');wi++;}
        // Oración
        const newSi=Math.min(Math.floor(wi/Math.max(words.length/Math.max(sents.length,1),1)),sents.length-1);
        if(newSi!==si){sents[si]?.classList.remove('sent-hi');si=newSi;sents[si]?.classList.add('sent-hi');}
        if(wi>=words.length)clearInterval(ht);
      },interval);
      src.onended=()=>{clearInterval(ht);words.forEach(w=>w.classList.remove('hi'));sents.forEach(s=>s.classList.remove('sent-hi'));resolve();};
    }else src.onended=resolve;
    src.start();setMSPlaying(true);
  });
}
